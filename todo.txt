  buildEntity(key: string, data: object) {
    let entity = {}; 
    let definition = Entities.find(item => item.type == key);

    for (const [sourceField, targetField] of Object.entries(definition.fields)) {
      if (data?.[targetField]) {
        entity[sourceField] = data?.[targetField];       
      }
    }

    return entity;
  }
